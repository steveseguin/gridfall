cmake_minimum_required(VERSION 3.21)
project(gridfall_ps1 LANGUAGES C CXX ASM)
file(GLOB ENGINE CONFIGURE_DEPENDS source/*.c)
psn00bsdk_add_executable(gridfall GPREL ${ENGINE} ps1/platform.c ps1/intro.c ps1/hud.c ps1/main.c ps1/memory.c)
target_include_directories(gridfall PRIVATE include source ps1 ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(gridfall PRIVATE GRIDFALL_PORT=1)
target_compile_options(gridfall PRIVATE -std=gnu99 -O2 -g -fwrapv -fno-strict-aliasing -fno-ipa-pure-const -ffunction-sections -fdata-sections -Wno-incompatible-pointer-types)
target_link_options(gridfall PRIVATE -Wl,--gc-sections)
psn00bsdk_add_cd_image(disc gridfall disc/iso.xml DEPENDS gridfall ${PROJECT_SOURCE_DIR}/build/GRIDFALL.DAT ${PROJECT_SOURCE_DIR}/build/SOUND.BNK)
